     1→'use client'
     2→
     3→import React, { useCallback, useEffect, useRef, useState } from 'react'
     4→import {
     5→  Plus,
     6→  Search,
     7→  FileSpreadsheet,
     8→  Download,
     9→  Pencil,
    10→  Trash2,
    11→  ChevronLeft,
    12→  ChevronRight,
    13→  GraduationCap,
    14→  Inbox,
    15→  Loader2,
    16→} from 'lucide-react'
    17→import { api } from '@/lib/api'
    18→import { Button } from '@/components/ui/button'
    19→import { Input } from '@/components/ui/input'
    20→import { Label } from '@/components/ui/label'
    21→import { Textarea } from '@/components/ui/textarea'
    22→import { Badge } from '@/components/ui/badge'
    23→import { Skeleton } from '@/components/ui/skeleton'
    24→import {
    25→  Select,
    26→  SelectContent,
    27→  SelectItem,
    28→  SelectTrigger,
    29→  SelectValue,
    30→} from '@/components/ui/select'
    31→import {
    32→  Dialog,

    34→  DialogDescription,
    35→  DialogFooter,
    36→  DialogHeader,
    37→  DialogTitle,
    38→} from '@/components/ui/dialog'
    39→import {
    40→  AlertDialog,
    41→  AlertDialogAction,

    46→  AlertDialogHeader,

    48→} from '@/components/ui/alert-dialog'
    49→import { toast } from 'sonner'

    51→// ── Types ────────────────────────────────────────────────────────────────────
    52→
    53→interface Siswa {
    54→  id: string
    55→  nis: string
    56→  nisn: string
    57→  nama: string
    58→  jenisKelamin: string
    59→  tempatLahir: string
    60→  tanggalLahir: string
    61→  agama: string
    62→  alamat: string
    63→  namaAyah: string
    64→  namaIbu: string
    65→  noHP: string
    66→  kelas: string
    67→  status: string
    68→}
    69→
    70→interface Kelas {
    71→  id: string
    72→  kodeKelas: string
    73→  namaKelas: string
    74→  waliKelas: string
    75→  status: string
    76→}
    77→
    78→interface SiswaResponse {
    79→  data: Siswa[]
    80→  total: number
    81→  page: number
    82→  limit: number
    83→  totalPages: number
    84→}
    85→
    86→interface SiswaApiRaw {
    87→  data: Siswa[]
    88→  pagination?: {
    89→    page: number
    90→    limit: number
    91→    total: number
    92→    totalPages: number
    93→  }
    94→}
    95→
    96→interface FormData {
    97→  nis: string
    98→  nisn: string
    99→  nama: string
   100→  jenisKelamin: string
   101→  tempatLahir: string
   102→  tanggalLahir: string
   103→  agama: string
   104→  alamat: string
   105→  namaAyah: string
   106→  namaIbu: string
   107→  noHP: string
   108→  kelas: string
   109→  status: string
   110→}

   113→  'Islam',

   115→  'Katolik',
   116→  'Hindu',
   117→  'Budha',
   118→  'Konghucu',
   119→]
   120→
   121→const EMPTY_FORM: FormData = {
   122→  nis: '',
   123→  nisn: '',
   124→  nama: '',
   125→  jenisKelamin: '',
   126→  tempatLahir: '',
   127→  tanggalLahir: '',
   128→  agama: '',
   129→  alamat: '',
   130→  namaAyah: '',
   131→  namaIbu: '',
   132→  noHP: '',
   133→  kelas: '',
   134→  status: 'aktif',
   135→}
   136→
   137→const PAGE_LIMIT = 10
   138→
   139→// ── Component ───────────────────────────────────────────────────────────────
   140→
   141→export default function DataSiswa() {
   142→  // ── Table state ──
   143→  const [siswaList, setSiswaList] = useState<Siswa[]>([])
   144→  const [total, setTotal] = useState(0)
   145→  const [totalPages, setTotalPages] = useState(1)
   146→  const [page, setPage] = useState(1)
   147→  const [search, setSearch] = useState('')
   148→  const [debouncedSearch, setDebouncedSearch] = useState('')
   149→  const [filterKelas, setFilterKelas] = useState('')
   150→  const [loading, setLoading] = useState(true)
   151→
   152→  // ── Kelas dropdown data ──
   153→  const [kelasList, setKelasList] = useState<Kelas[]>([])
   154→
   155→  // ── Modal state ──
   156→  const [modalOpen, setModalOpen] = useState(false)
   157→  const [editingId, setEditingId] = useState<string | null>(null)
   158→  const [form, setForm] = useState<FormData>(EMPTY_FORM)
   159→  const [formSubmitting, setFormSubmitting] = useState(false)
   160→
   161→  // ── Delete confirmation state ──
   162→  const [deleteOpen, setDeleteOpen] = useState(false)
   163→  const [deletingItem, setDeletingItem] = useState<Siswa | null>(null)
   164→  const [deleting, setDeleting] = useState(false)
   165→
   166→  // ── Refs ──
   167→  const fileInputRef = useRef<HTMLInputElement>(null)
   168→  const debounceTimer = useRef<ReturnType<typeof setTimeout>>()
   169→
   170→  // ── Fetch kelas list ──
   171→  const fetchKelas = useCallback(async () => {
   172→    try {
   173→      const res = await api.get<Kelas[]>('/kelas')
   174→      setKelasList(Array.isArray(res) ? res : (res as unknown as { data: Kelas[] }).data ?? [])
   175→    } catch {
   176→      // silently fail, dropdown will just be empty
   177→    }
   178→  }, [])
   179→
   180→  // ── Fetch siswa list ──
   181→  const fetchSiswa = useCallback(async () => {
   182→    setLoading(true)
   183→    try {
   184→      const params = new URLSearchParams()
   185→      params.set('page', String(page))
   186→      params.set('limit', String(PAGE_LIMIT))
   187→      if (debouncedSearch) params.set('search', debouncedSearch)
   188→      if (filterKelas) params.set('kelas', filterKelas)
   189→
   190→      const res = await api.get<SiswaApiRaw>(`/siswa?${params.toString()}`)
   191→      const raw = res as SiswaApiRaw
   192→      const data = raw.data ?? []
   193→      const pag = raw.pagination
   194→      setSiswaList(data)
   195→      setTotal(pag?.total ?? 0)
   196→      setTotalPages(pag?.totalPages ?? 1)
   197→      setPage(pag?.page ?? 1)
   198→    } catch {
   199→      toast.error('Gagal memuat data siswa')
   200→    } finally {
   201→      setLoading(false)
   202→    }
   203→  }, [page, debouncedSearch, filterKelas])
   204→
   205→  // ── Debounced search ──
   206→  useEffect(() => {
   207→    if (debounceTimer.current) clearTimeout(debounceTimer.current)
   208→    debounceTimer.current = setTimeout(() => {
   209→      setDebouncedSearch(search)
   210→      setPage(1)
   211→    }, 400)
   212→    return () => {
   213→      if (debounceTimer.current) clearTimeout(debounceTimer.current)
   214→    }
   215→  }, [search])
   216→
   217→  // ── Data fetching effects ──
   218→  useEffect(() => {
   219→    fetchKelas()
   220→  }, [fetchKelas])
   221→
   222→  useEffect(() => {
   223→    fetchSiswa()
   224→  }, [fetchSiswa])
   225→
   226→  // ── Helpers ──
   227→  const getKelasNama = (kodeKelas: string) => {
   228→    const k = kelasList.find((k) => k.kodeKelas === kodeKelas)
   229→    return k?.namaKelas ?? kodeKelas
   230→  }
   231→
   232→  const openAddModal = () => {
   233→    setEditingId(null)
   234→    setForm(EMPTY_FORM)
   235→    setModalOpen(true)
   236→  }
   237→
   238→  const openEditModal = (siswa: Siswa) => {
   239→    setEditingId(siswa.id)
   240→    setForm({
   241→      nis: siswa.nis,
   242→      nisn: siswa.nisn,
   243→      nama: siswa.nama,
   244→      jenisKelamin: siswa.jenisKelamin,
   245→      tempatLahir: siswa.tempatLahir,
   246→      tanggalLahir: siswa.tanggalLahir,
   247→      agama: siswa.agama,
   248→      alamat: siswa.alamat,
   249→      namaAyah: siswa.namaAyah,
   250→      namaIbu: siswa.namaIbu,
   251→      noHP: siswa.noHP ?? '',
   252→      kelas: siswa.kelas || '',
   253→      status: siswa.status,
   254→    })
   255→    setModalOpen(true)
   256→  }
   257→
   258→  const handleFormChange = (field: keyof FormData, value: string) => {
   259→    setForm((prev) => ({ ...prev, [field]: value }))
   260→  }
   261→
   262→  const handleSubmit = async () => {
   263→    if (!form.nis.trim() || !form.nisn.trim() || !form.nama.trim()) {
   264→      toast.error('NIS, NISN, dan Nama wajib diisi')
   265→      return
   266→    }
   267→    if (!form.kelas) {
   268→      toast.error('Kelas wajib dipilih')
   269→      return
   270→    }
   271→
   272→    setFormSubmitting(true)
   273→    try {
   274→      const payload = {
   275→        nis: form.nis,
   276→        nisn: form.nisn,
   277→        nama: form.nama,
   278→        jenisKelamin: form.jenisKelamin,
   279→        tempatLahir: form.tempatLahir,
   280→        tanggalLahir: form.tanggalLahir,
   281→        agama: form.agama,
   282→        alamat: form.alamat,
   283→        namaAyah: form.namaAyah,
   284→        namaIbu: form.namaIbu,
   285→        noHP: form.noHP,
   286→        kelas: form.kelas,
   287→        status: form.status,
   288→      }
   289→      if (editingId) {
   290→        await api.put('/siswa', { id: editingId, ...payload })
   291→        toast.success('Data siswa berhasil diperbarui')
   292→      } else {
   293→        await api.post('/siswa', payload)
   294→        toast.success('Data siswa berhasil ditambahkan')
   295→      }
   296→      setModalOpen(false)
   297→      fetchSiswa()
   298→    } catch (err) {
   299→      toast.error(err instanceof Error ? err.message : 'Gagal menyimpan data')
   300→    } finally {
   301→      setFormSubmitting(false)
   302→    }
   303→  }
   304→
   305→  const confirmDelete = (siswa: Siswa) => {
   306→    setDeletingItem(siswa)
   307→    setDeleteOpen(true)
   308→  }
   309→
   310→  const handleDelete = async () => {
   311→    if (!deletingItem) return
   312→    setDeleting(true)
   313→    try {
   314→      await api.del(`/siswa?id=${deletingItem.id}`)
   315→      toast.success('Data siswa berhasil dihapus')
   316→      setDeleteOpen(false)
   317→      setDeletingItem(null)
   318→      fetchSiswa()
   319→    } catch (err) {
   320→      toast.error(err instanceof Error ? err.message : 'Gagal menghapus data')
   321→    } finally {
   322→      setDeleting(false)
   323→    }
   324→  }
   325→
   326→  const handleImportExcel = () => {
   327→    fileInputRef.current?.click()
   328→  }
   329→
   330→  const handleFileChange = async (e: React.ChangeEvent<HTMLInputElement>) => {
   331→    const file = e.target.files?.[0]
   332→    if (!file) return
   333→    const formData = new FormData()
   334→    formData.append('file', file)
   335→    try {
   336→      const token = (await import('@/lib/store')).useAppStore.getState().token
   337→      const headers: Record<string, string> = {}
   338→      if (token) headers['Authorization'] = `Bearer ${token}`
   339→      const res = await fetch('/api/siswa/import', {
   340→        method: 'POST',
   341→        headers,
   342→        body: formData,
   343→      })
   344→      const json = await res.json()
   345→      if (!res.ok) throw new Error(json.error || 'Gagal import')
   346→      toast.success('Data berhasil diimport')
   347→      fetchSiswa()
   348→    } catch (err) {
   349→      toast.error(err instanceof Error ? err.message : 'Gagal mengimport file')
   350→    } finally {
   351→      if (fileInputRef.current) fileInputRef.current.value = ''
   352→    }
   353→  }
   354→
   355→  const handleExportExcel = async () => {
   356→    try {
   357→      const token = (await import('@/lib/store')).useAppStore.getState().token
   358→      const headers: Record<string, string> = {}
   359→      if (token) headers['Authorization'] = `Bearer ${token}`
   360→      const params = new URLSearchParams()
   361→      if (debouncedSearch) params.set('search', debouncedSearch)
   362→      if (filterKelas) params.set('kelas', filterKelas)
   363→      const res = await fetch(`/api/siswa/export?${params.toString()}`, { headers })
   364→      if (!res.ok) throw new Error('Gagal export')
   365→      const blob = await res.blob()
   366→      const url = URL.createObjectURL(blob)
   367→      const a = document.createElement('a')
   368→      a.href = url
   369→      a.download = 'data-siswa.xlsx'
   370→      a.click()
   371→      URL.revokeObjectURL(url)
   372→      toast.success('Data berhasil diexport')
   373→    } catch (err) {
   374→      toast.error(err instanceof Error ? err.message : 'Gagal mengexport data')
   375→    }
   376→  }
   377→
   378→  // ── Pagination range ──
   379→  const startIdx = (page - 1) * PAGE_LIMIT
   380→  const endIdx = Math.min(startIdx + PAGE_LIMIT, total)
   381→
   382→  return (
   383→    <section className="w-full" style={{ backgroundColor: '#f4f6f9' }}>
   384→      <div className="mx-auto max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   385→        {/* ── Header ── */}
   386→        <div className="mb-6">
   387→          <h1
   388→            className="text-2xl font-bold"
   389→            style={{ color: '#0a2540' }}
   390→          >
   391→            Data Siswa
   392→          </h1>
   393→          <p className="mt-1 text-sm" style={{ color: '#0a2540' }}>
   394→            Kelola data siswa sekolah dengan mudah
   395→          </p>
   396→        </div>
   397→
   398→        {/* ── Card ── */}
   399→        <div className="rounded-2xl bg-white p-4 shadow-sm sm:p-6">
   400→          {/* ── Action Bar ── */}
   401→          <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   402→            {/* Left: buttons */}
   403→            <div className="flex flex-wrap items-center gap-2">
   404→              <Button
   405→                onClick={openAddModal}
   406→                className="gap-1.5 text-white"
   407→                style={{ backgroundColor: '#10b981' }}
   408→              >
   409→                <Plus className="size-4" />
   410→                Tambah
   411→              </Button>
   412→              <Button
   413→                variant="outline"
   414→                onClick={handleImportExcel}
   415→                className="gap-1.5"
   416→                style={{ color: '#0a2540', borderColor: '#d1d5db' }}
   417→              >
   418→                <FileSpreadsheet className="size-4" />
   419→                Import Excel
   420→              </Button>
   421→              <Button
   422→                variant="outline"
   423→                onClick={handleExportExcel}
   424→                className="gap-1.5"
   425→                style={{ color: '#0a2540', borderColor: '#d1d5db' }}
   426→              >
   427→                <Download className="size-4" />
   428→                Export Excel
   429→              </Button>
   430→              <input
   431→                ref={fileInputRef}
   432→                type="file"
   433→                accept=".xlsx,.xls"
   434→                className="hidden"
   435→                onChange={handleFileChange}
   436→              />
   437→            </div>
   438→
   439→            {/* Right: search + filter */}
   440→            <div className="flex flex-wrap items-center gap-2">
   441→              <div className="relative">
   442→                <Search
   443→                  className="absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-gray-400"
   444→                />
   445→                <Input
   446→                  placeholder="Cari siswa..."
   447→                  value={search}
   448→                  onChange={(e) => setSearch(e.target.value)}
   449→                  className="w-56 pl-9"
   450→                />
   451→              </div>
   452→              <Select value={filterKelas} onValueChange={(v) => { setFilterKelas(v); setPage(1) }}>
   453→                <SelectTrigger className="w-44">
   454→                  <SelectValue placeholder="Semua Kelas" />
   455→                </SelectTrigger>
   456→                <SelectContent>
   457→                  <SelectItem value="__all__">Semua Kelas</SelectItem>
   458→                  {kelasList.map((k) => (
   459→                    <SelectItem key={k.kodeKelas} value={k.kodeKelas}>
   460→                      {k.namaKelas}
   461→                    </SelectItem>
   462→                  ))}
   463→                </SelectContent>
   464→              </Select>
   465→            </div>
   466→          </div>
   467→
   468→          {/* ── Table ── */}
   469→          <div className="overflow-x-auto">
   470→            {loading ? (
   471→              /* Skeleton */
   472→              <div className="space-y-3">
   473→                {Array.from({ length: 5 }).map((_, i) => (
   474→                  <div key={i} className="flex items-center gap-4">
   475→                    <Skeleton className="h-8 w-8" />
   476→                    <Skeleton className="h-8 w-20" />
   477→                    <Skeleton className="h-8 w-24" />
   478→                    <Skeleton className="h-8 flex-1" />
   479→                    <Skeleton className="h-8 w-12" />
   480→                    <Skeleton className="h-8 w-20" />
   481→                    <Skeleton className="h-8 w-16" />
   482→                    <Skeleton className="h-8 w-20" />
   483→                  </div>
   484→                ))}
   485→              </div>
   486→            ) : siswaList.length === 0 ? (
   487→              /* Empty state */
   488→              <div className="flex flex-col items-center justify-center py-16 text-center">
   489→                <div
   490→                  className="mb-4 flex size-16 items-center justify-center rounded-full"
   491→                  style={{ backgroundColor: '#f0fdf4' }}
   492→                >
   493→                  <Inbox className="size-8" style={{ color: '#10b981' }} />
   494→                </div>
   495→                <h3
   496→                  className="text-base font-semibold"
   497→                  style={{ color: '#0a2540' }}
   498→                >
   499→                  Data Kosong
   500→                </h3>
   501→                <p className="mt-1 max-w-xs text-sm text-gray-500">
   502→                  {debouncedSearch || filterKelas
   503→                    ? 'Tidak ditemukan siswa yang sesuai dengan filter.'
   504→                    : 'Belum ada data siswa. Klik tombol Tambah untuk menambahkan siswa baru.'}
   505→                </p>
   506→              </div>
   507→            ) : (
   508→              <table className="siadak-table w-full text-sm">
   509→                <thead>
   510→                  <tr
   511→                    className="border-b text-left text-xs font-semibold uppercase tracking-wide"
   512→                    style={{ color: '#0a2540', backgroundColor: '#f9fafb' }}
   513→                  >
   514→                    <th className="px-3 py-3">No</th>
   515→                    <th className="px-3 py-3">NIS</th>
   516→                    <th className="px-3 py-3">NISN</th>
   517→                    <th className="px-3 py-3">Nama</th>
   518→                    <th className="px-3 py-3">JK</th>
   519→                    <th className="px-3 py-3">Kelas</th>
   520→                    <th className="px-3 py-3">Status</th>
   521→                    <th className="px-3 py-3 text-right">Aksi</th>
   522→                  </tr>
   523→                </thead>
   524→                <tbody>
   525→                  {siswaList.map((siswa, idx) => (
   526→                    <tr
   527→                      key={siswa.id}
   528→                      className="border-b transition-colors last:border-b-0 hover:bg-gray-50"
   529→                    >
   530→                      <td className="px-3 py-3" style={{ color: '#0a2540' }}>
   531→                        {startIdx + idx + 1}
   532→                      </td>
   533→                      <td className="px-3 py-3" style={{ color: '#0a2540' }}>
   534→                        {siswa.nis}
   535→                      </td>
   536→                      <td className="px-3 py-3" style={{ color: '#0a2540' }}>
   537→                        {siswa.nisn}
   538→                      </td>
   539→                      <td
   540→                        className="px-3 py-3 font-medium"
   541→                        style={{ color: '#0a2540' }}
   542→                      >
   543→                        {siswa.nama}
   544→                      </td>
   545→                      <td className="px-3 py-3">
   546→                        <Badge
   547→                          className={
   548→                            siswa.jenisKelamin?.startsWith('L')
   549→                              ? 'bg-blue-100 text-blue-700'
   550→                              : 'bg-pink-100 text-pink-700'
   551→                          }
   552→                        >
   553→                          {siswa.jenisKelamin?.startsWith('L') ? 'L' : 'P'}
   554→                        </Badge>
   555→                      </td>
   556→                      <td className="px-3 py-3" style={{ color: '#0a2540' }}>
   557→                        {getKelasNama(siswa.kelas)}
   558→                      </td>
   559→                      <td className="px-3 py-3">
   560→                        {siswa.status === 'aktif' ? (
   561→                          <Badge className="bg-green-100 text-green-700">
   562→                            Aktif
   563→                          </Badge>
   564→                        ) : (
   565→                          <Badge className="bg-red-100 text-red-700">
   566→                            Nonaktif
   567→                          </Badge>
   568→                        )}
   569→                      </td>
   570→                      <td className="px-3 py-3 text-right">
   571→                        <div className="flex items-center justify-end gap-1">
   572→                          <button
   573→                            onClick={() => openEditModal(siswa)}
   574→                            className="rounded-lg p-1.5 transition-colors hover:bg-blue-50"
   575→                            title="Edit"
   576→                          >
   577→                            <Pencil className="size-4" style={{ color: '#2563eb' }} />
   578→                          </button>
   579→                          <button
   580→                            onClick={() => confirmDelete(siswa)}
   581→                            className="rounded-lg p-1.5 transition-colors hover:bg-red-50"
   582→                            title="Hapus"
   583→                          >
   584→                            <Trash2 className="size-4" style={{ color: '#ef4444' }} />
   585→                          </button>
   586→                        </div>
   587→                      </td>
   588→                    </tr>

   625→                      <span
   626→                        key={`ellipsis-${i}`}

   629→                        ...
   630→                      </span>

   632→                      <button
   633→                        key={item}
   634→                        onClick={() => setPage(item)}

   636→                          'flex size-8 items-center justify-center rounded-lg text-sm font-medium transition-colors ' +

   642→                          page === item

   645→                        }
   646→                      >
   647→                        {item}
   648→                      </button>
   649→                    ),

   656→                  <ChevronRight className="size-4" />
   657→                </button>
