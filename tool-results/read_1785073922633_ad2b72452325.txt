   154→
   155→function getPredikatFromRataRata(rata: number): string {
   156→  if (rata >= 90) return 'A'
   157→  if (rata >= 80) return 'B'
   158→  if (rata >= 70) return 'C'
   159→  return 'D'
   160→}
   161→
   162→function getPredikatBadge(predikat: string) {
   163→  const colors: Record<string, string> = {
   164→    A: 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-0',
   165→    B: 'bg-blue-100 text-blue-700 hover:bg-blue-100 border-0',
   166→    C: 'bg-amber-100 text-amber-700 hover:bg-amber-100 border-0',
   167→    D: 'bg-red-100 text-red-600 hover:bg-red-100 border-0',
   168→  }
   169→  return (
   170→    <Badge className={colors[predikat] || 'bg-gray-100 text-gray-600 hover:bg-gray-100 border-0'}>
   171→      {predikat}
   172→    </Badge>
   173→  )
   174→}
   175→
   176→function getStatusLulusBadge(lulus: boolean) {
   177→  if (lulus) {
   178→    return (
   179→      <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-0">
   180→        Lulus
   181→      </Badge>
   182→    )
   183→  }
   184→  return (
   185→    <Badge className="bg-red-100 text-red-600 hover:bg-red-100 border-0">
   186→      Tidak Lulus
   187→    </Badge>
   188→  )
   189→}
   190→
   191→function getPersentaseColor(pct: number): string {
   192→  if (pct >= 80) return 'text-emerald-600'
   193→  if (pct >= 60) return 'text-amber-600'
   194→  return 'text-red-600'
   195→}

   201→export default function Laporan() {
   202→  // ── Report type ──
   203→  const [tipe, setTipe] = useState<TipeLaporan>('nilai')
   204→
   205→  // ── Nilai filter state ──
   206→  const [tahunAjaran, setTahunAjaran] = useState('2024/2025')
   207→  const [semester, setSemester] = useState('Ganjil')
   208→  const [nilaiKelas, setNilaiKelas] = useState('')
   209→  const [nilaiMapel, setNilaiMapel] = useState('all')
   210→
   211→  // ── Absensi filter state ──
   212→  const [absensiKelas, setAbsensiKelas] = useState('')
   213→  const [tanggalMulai, setTanggalMulai] = useState('')
   214→  const [tanggalSelesai, setTanggalSelesai] = useState('')
   215→
   216→  // ── Dropdown data ──
   217→  const [kelasList, setKelasList] = useState<KelasItem[]>([])
   218→  const [mapelList, setMapelList] = useState<MapelItem[]>([])
   219→  const [loadingDropdowns, setLoadingDropdowns] = useState(true)
   220→
   221→  // ── Result state ──
   222→  const [loading, setLoading] = useState(false)
   223→  const [hasSearched, setHasSearched] = useState(false)
   224→  const [nilaiResult, setNilaiResult] = useState<NilaiLaporanResponse | null>(null)
   225→  const [absensiResult, setAbsensiResult] = useState<AbsensiLaporanResponse | null>(null)
   226→
   227→  // ── Fetch kelas & mapel dropdowns ──
   228→  const fetchDropdowns = useCallback(async () => {
   229→    setLoadingDropdowns(true)
   230→    try {
   231→      const [kelasRes, mapelRes] = await Promise.all([
   232→        api.get<KelasResponse>('/kelas'),
   233→        api.get<MapelResponse>('/mapel'),
   234→      ])
   235→      const kelasData = (kelasRes as KelasResponse).data
   236→      const mapelData = (mapelRes as MapelResponse).data
   237→      setKelasList(Array.isArray(kelasData) ? kelasData : [])
   238→      setMapelList(Array.isArray(mapelData) ? mapelData : [])
   239→    } catch {
   240→      toast.error('Gagal memuat data referensi')
   241→    } finally {
   242→      setLoadingDropdowns(false)
   243→    }
   244→  }, [])
   245→
   246→  useEffect(() => {
   247→    fetchDropdowns()
   248→  }, [fetchDropdowns])
   249→
   250→  // ── Reset results on type change ──
   251→  useEffect(() => {
   252→    setHasSearched(false)
   253→    setNilaiResult(null)
   254→    setAbsensiResult(null)
   255→  }, [tipe])
   256→
   257→  // ── Validate & fetch ──
   258→  const handleTampilkan = useCallback(async () => {
   259→    if (tipe === 'nilai') {
   260→      if (!nilaiKelas) {
   261→        toast.error('Pilih Kelas terlebih dahulu')
   262→        return
   263→      }
   264→      setLoading(true)
   265→      setHasSearched(true)
   266→      try {
   267→        const params = new URLSearchParams({
   268→          tipe: 'nilai',
   269→          kelas: nilaiKelas,
   270→          ...(nilaiMapel && nilaiMapel !== 'all' ? { mapel: nilaiMapel } : {}),
   271→          semester,
   272→          tahunAjaran,
   273→        })
   274→        const res = await api.get<NilaiLaporanResponse>(`/laporan?${params.toString()}`)
   275→        setNilaiResult(res as NilaiLaporanResponse)
   276→      } catch {
   277→        toast.error('Gagal memuat laporan nilai')
   278→        setNilaiResult(null)
   279→      } finally {
   280→        setLoading(false)
   281→      }
   282→    } else {
   283→      if (!absensiKelas) {
   284→        toast.error('Pilih Kelas terlebih dahulu')
   285→        return
   286→      }
   287→      if (!tanggalMulai || !tanggalSelesai) {
   288→        toast.error('Tanggal Mulai dan Tanggal Selesai wajib diisi')
   289→        return
   290→      }
   291→      if (tanggalMulai > tanggalSelesai) {
   292→        toast.error('Tanggal Mulai tidak boleh lebih dari Tanggal Selesai')
   293→        return
   294→      }
   295→      setLoading(true)
   296→      setHasSearched(true)
   297→      try {
   298→        const params = new URLSearchParams({
   299→          tipe: 'absensi',
   300→          kelas: absensiKelas,
   301→          tanggalMulai,
   302→          tanggalSelesai,
   303→        })
   304→        const res = await api.get<AbsensiLaporanResponse>(`/laporan?${params.toString()}`)
   305→        setAbsensiResult(res as AbsensiLaporanResponse)
   306→      } catch {
   307→        toast.error('Gagal memuat laporan absensi')
   308→        setAbsensiResult(null)
   309→      } finally {
   310→        setLoading(false)
   311→      }
   312→    }
   313→  }, [tipe, tahunAjaran, semester, nilaiKelas, nilaiMapel, absensiKelas, tanggalMulai, tanggalSelesai])
   314→
   315→  // ── Export Excel (CSV) ──
   316→  const handleExportExcel = useCallback(() => {
   317→    if (tipe === 'nilai') {
   318→      if (!nilaiResult || nilaiResult.data.length === 0) {
   319→        toast.error('Tidak ada data untuk diekspor')
   320→        return
   321→      }
   322→      const headers = ['No', 'NIS', 'Nama', 'Kelas', 'Jumlah Mapel', 'Rata-Rata', 'Nilai Tertinggi', 'Nilai Terendah', 'Mapel Lulus', 'Mapel Tidak Lulus', 'Predikat', 'Status Lulus']
   323→      const rows = nilaiResult.data.map((item, idx) => [
   324→        idx + 1,
   325→        item.nis,
   326→        item.nama,
   327→        item.kelas,
   328→        item.jumlahMapel,
   329→        item.rataRata,
   330→        item.nilaiTertinggi,
   331→        item.nilaiTerendah,
   332→        item.mapelLulus,
   333→        item.mapelTidakLulus,
   334→        getPredikatFromRataRata(item.rataRata),
   335→        item.mapelTidakLulus === 0 ? 'Lulus' : 'Tidak Lulus',
   336→      ])
   337→      const csvContent = [
   338→        headers.join(','),
   339→        ...rows.map((row) =>
   340→          row.map((cell) => {
   341→            const str = String(cell)
   342→            return str.includes(',') || str.includes('"') || str.includes('\n')
   343→              ? `"${str.replace(/"/g, '""')}"`
   344→              : str
   345→          }).join(','),
   346→        ),
   347→      ].join('\n')
   348→      const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   349→      const url = URL.createObjectURL(blob)
   350→      const link = document.createElement('a')
   351→      link.href = url
   352→      link.download = `Laporan_Nilai_${nilaiKelas}_${semester}_${tahunAjaran.replace('/', '-')}.csv`
   353→      document.body.appendChild(link)
   354→      link.click()
   355→      document.body.removeChild(link)
   356→      URL.revokeObjectURL(url)
   357→    } else {
   358→      if (!absensiResult || absensiResult.data.length === 0) {
   359→        toast.error('Tidak ada data untuk diekspor')
   360→        return
   361→      }
   362→      const headers = ['No', 'NIS', 'Nama', 'Kelas', 'Hadir', 'Sakit', 'Izin', 'Alpha', 'Total', 'Persentase Kehadiran (%)']
   363→      const rows = absensiResult.data.map((item, idx) => [
   364→        idx + 1,
   365→        item.nis,
   366→        item.nama,
   367→        item.kelas,
   368→        item.hadir,
   369→        item.sakit,
   370→        item.izin,
   371→        item.alpha,
   372→        item.totalHari,
   373→        item.persentaseKehadiran,
   374→      ])
   375→      const csvContent = [
   376→        headers.join(','),
   377→        ...rows.map((row) =>
   378→          row.map((cell) => {
   379→            const str = String(cell)
   380→            return str.includes(',') || str.includes('"') || str.includes('\n')
   381→              ? `"${str.replace(/"/g, '""')}"`
   382→              : str
   383→          }).join(','),
   384→        ),
   385→      ].join('\n')
   386→      const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   387→      const url = URL.createObjectURL(blob)
   388→      const link = document.createElement('a')
   389→      link.href = url
   390→      link.download = `Laporan_Absensi_${absensiKelas}_${tanggalMulai}_sd_${tanggalSelesai}.csv`
   391→      document.body.appendChild(link)
   392→      link.click()
   393→      document.body.removeChild(link)
   394→      URL.revokeObjectURL(url)
   395→    }
   396→    toast.success('Data berhasil diekspor ke Excel')
   397→  }, [tipe, nilaiResult, absensiResult, nilaiKelas, semester, tahunAjaran, absensiKelas, tanggalMulai, tanggalSelesai])
   398→
   399→  // ── Export PDF (placeholder) ──
   400→  const handleExportPDF = useCallback(() => {
   401→    toast.info('Fitur export PDF akan segera tersedia. Gunakan Export Excel sebagai alternatif.')
   402→  }, [])
   403→
   404→  // ── Derived data ──
   405→  const nilaiData = nilaiResult?.data ?? []
   406→  const absensiData = absensiResult?.data ?? []
   407→  const hasData = tipe === 'nilai' ? nilaiData.length > 0 : absensiData.length > 0
   408→
   409→  // ── Render ──
   410→  return (
   411→    <div className="space-y-6 animate-fadeIn">
   412→      {/* ── Report Type Selector ── */}
   413→      <div className="bg-white rounded-2xl shadow-sm p-4 sm:p-6">
   414→        <div className="flex items-center gap-2 mb-4">
   415→          <BarChart3 className="h-5 w-5 text-[#2563eb]" />
   416→          <h2 className="text-lg font-semibold text-foreground">Laporan SIAKAD</h2>
   417→        </div>
   418→
   419→        <RadioGroup
   420→          value={tipe}
   421→          onValueChange={(v) => setTipe(v as TipeLaporan)}
   422→          className="flex flex-col sm:flex-row gap-3"
   423→        >
   424→          <label
   425→            htmlFor="tipe-nilai"
   426→            className={[
   427→              'flex items-center gap-3 px-5 py-3 rounded-xl border-2 cursor-pointer transition-all',
   428→              tipe === 'nilai'
   429→                ? 'border-[#2563eb] bg-blue-50'
   430→                : 'border-border bg-white hover:border-muted-foreground/30',
   431→            ].join(' ')}
   432→          >
   433→            <RadioGroupItem value="nilai" id="tipe-nilai" />
   434→            <div className="flex items-center gap-2">
   435→              <BarChart3 className={`h-5 w-5 ${tipe === 'nilai' ? 'text-[#2563eb]' : 'text-muted-foreground'}`} />
   436→              <span className={`font-medium ${tipe === 'nilai' ? 'text-[#2563eb]' : 'text-muted-foreground'}`}>
   437→                Laporan Nilai
   438→              </span>
   439→            </div>
   440→          </label>
   441→
   442→          <label
   443→            htmlFor="tipe-absensi"
   444→            className={[
   445→              'flex items-center gap-3 px-5 py-3 rounded-xl border-2 cursor-pointer transition-all',
   446→              tipe === 'absensi'
   447→                ? 'border-[#2563eb] bg-blue-50'
   448→                : 'border-border bg-white hover:border-muted-foreground/30',
   449→            ].join(' ')}
   450→          >
   451→            <RadioGroupItem value="absensi" id="tipe-absensi" />
   452→            <div className="flex items-center gap-2">
   453→              <CalendarDays className={`h-5 w-5 ${tipe === 'absensi' ? 'text-[#2563eb]' : 'text-muted-foreground'}`} />
   454→              <span className={`font-medium ${tipe === 'absensi' ? 'text-[#2563eb]' : 'text-muted-foreground'}`}>
   455→                Laporan Absensi
   456→              </span>
   457→            </div>
   458→          </label>
   459→        </RadioGroup>
   460→      </div>
   461→
   462→      {/* ── Filter Card ── */}
   463→      <div className="bg-white rounded-2xl shadow-sm p-4 sm:p-6">
   464→        <div className="flex items-center gap-2 mb-4">
   465→          <Filter className="h-5 w-5 text-[#2563eb]" />
   466→          <h2 className="text-lg font-semibold text-foreground">
   467→            Filter {tipe === 'nilai' ? 'Laporan Nilai' : 'Laporan Absensi'}
   468→          </h2>
   469→        </div>
   470→
   471→        {tipe === 'nilai' ? (
   472→          /* ── Nilai Filters ── */
   473→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-5 gap-3">
   474→            <div className="space-y-1.5">
   475→              <Label className="text-xs font-medium text-muted-foreground">Tahun Ajaran</Label>
   476→              <Select value={tahunAjaran} onValueChange={setTahunAjaran}>
   477→                <SelectTrigger className="w-full">
   478→                  <SelectValue placeholder="Pilih Tahun Ajaran" />
   479→                </SelectTrigger>
   480→                <SelectContent>
   481→                  {TAHUN_AJARAN_OPTIONS.map((ta) => (
   482→                    <SelectItem key={ta} value={ta}>
   483→                      {ta}
   484→                    </SelectItem>
   485→                  ))}
   486→                </SelectContent>
   487→              </Select>
   488→            </div>
   489→
   490→            <div className="space-y-1.5">
   491→              <Label className="text-xs font-medium text-muted-foreground">Semester</Label>
   492→              <Select value={semester} onValueChange={setSemester}>
   493→                <SelectTrigger className="w-full">
   494→                  <SelectValue placeholder="Pilih Semester" />
   495→                </SelectTrigger>
   496→                <SelectContent>
   497→                  {SEMESTER_OPTIONS.map((s) => (
   498→                    <SelectItem key={s} value={s}>
   499→                      {s}
   500→                    </SelectItem>
   501→                  ))}
   502→                </SelectContent>
   503→              </Select>
   504→            </div>
   505→
   506→            <div className="space-y-1.5">
   507→              <Label className="text-xs font-medium text-muted-foreground">Kelas</Label>

   509→                <SelectTrigger className="w-full" disabled={loadingDropdowns}>
   510→                  <SelectValue placeholder={loadingDropdowns ? 'Memuat...' : 'Pilih Kelas'} />
   511→                </SelectTrigger>
   512→                <SelectContent>
   513→                  {kelasList.map((k) => (
   514→                    <SelectItem key={k.kodeKelas} value={k.kodeKelas}>
   515→                      {k.namaKelas}
   516→                    </SelectItem>
   517→                  ))}
   518→                </SelectContent>
   519→              </Select>
   520→            </div>
   521→
   522→            <div className="space-y-1.5">
   523→              <Label className="text-xs font-medium text-muted-foreground">Mata Pelajaran</Label>
   524→              <Select value={nilaiMapel} onValueChange={setNilaiMapel}>

   526→                  <SelectValue placeholder={loadingDropdowns ? 'Memuat...' : 'Semua Mapel'} />
   527→                </SelectTrigger>
   528→                <SelectContent>
   529→                  <SelectItem value="all">Semua Mapel</SelectItem>
   530→                  {mapelList.map((m) => (
   531→                    <SelectItem key={m.kodeMapel} value={m.kodeMapel}>
   532→                      {m.namaMapel}

   539→            <div className="space-y-1.5 flex items-end">
