   127→   127→// ── Component ───────────────────────────────────────────────────────────────
   128→   128→
   129→   129→export default function DataSiswa() {
   130→   130→  // ── Table state ──
   131→   131→  const [siswaList, setSiswaList] = useState<Siswa[]>([])
   132→   132→  const [total, setTotal] = useState(0)
   133→   133→  const [totalPages, setTotalPages] = useState(1)
   134→   134→  const [page, setPage] = useState(1)
   135→   135→  const [search, setSearch] = useState('')
   136→   136→  const [debouncedSearch, setDebouncedSearch] = useState('')
   137→   137→  const [filterKelas, setFilterKelas] = useState('')
   138→   138→  const [loading, setLoading] = useState(true)
   139→   139→
   140→   140→  // ── Kelas dropdown data ──
   141→   141→  const [kelasList, setKelasList] = useState<Kelas[]>([])
   142→   142→
   143→   143→  // ── Modal state ──
   144→   144→  const [modalOpen, setModalOpen] = useState(false)
   145→   145→  const [editingId, setEditingId] = useState<number | null>(null)
   146→   146→  const [form, setForm] = useState<FormData>(EMPTY_FORM)
   147→   147→  const [formSubmitting, setFormSubmitting] = useState(false)
   148→   148→
   149→   149→  // ── Delete confirmation state ──
   150→   150→  const [deleteOpen, setDeleteOpen] = useState(false)
   151→   151→  const [deletingItem, setDeletingItem] = useState<Siswa | null>(null)
   152→   152→  const [deleting, setDeleting] = useState(false)
   153→   153→
   154→   154→  // ── Refs ──
   155→   155→  const fileInputRef = useRef<HTMLInputElement>(null)
   156→   156→  const debounceTimer = useRef<ReturnType<typeof setTimeout>>()
   157→   157→
   158→   158→  // ── Fetch kelas list ──
   159→   159→  const fetchKelas = useCallback(async () => {
   160→   160→    try {
   161→   161→      const res = await api.get<Kelas[]>('/kelas')
   162→   162→      setKelasList(Array.isArray(res) ? res : (res as unknown as { data: Kelas[] }).data ?? [])
   163→   163→    } catch {
   164→   164→      // silently fail, dropdown will just be empty
   165→   165→    }
   166→   166→  }, [])
   167→   167→
   168→   168→  // ── Fetch siswa list ──
   169→   169→  const fetchSiswa = useCallback(async () => {
   170→   170→    setLoading(true)
   171→   171→    try {
   172→   172→      const params = new URLSearchParams()
   173→   173→      params.set('page', String(page))
   174→   174→      params.set('limit', String(PAGE_LIMIT))
   175→   175→      if (debouncedSearch) params.set('search', debouncedSearch)
   176→   176→      if (filterKelas) params.set('kelas', filterKelas)
   177→   177→
   178→   178→      const res = await api.get<SiswaResponse>(`/siswa?${params.toString()}`)
   179→   179→      const data = res as SiswaResponse
   180→   180→      setSiswaList(data.data ?? [])
   181→   181→      setTotal(data.total ?? 0)
   182→   182→      setTotalPages(data.totalPages ?? 1)
   183→   183→    } catch {
   184→   184→      toast.error('Gagal memuat data siswa')
   185→   185→    } finally {
   186→   186→      setLoading(false)
   187→   187→    }
   188→   188→  }, [page, debouncedSearch, filterKelas])
   189→   189→
   190→   190→  // ── Debounced search ──
   191→   191→  useEffect(() => {
   192→   192→    if (debounceTimer.current) clearTimeout(debounceTimer.current)
   193→   193→    debounceTimer.current = setTimeout(() => {
   194→   194→      setDebouncedSearch(search)
   195→   195→      setPage(1)
   196→   196→    }, 400)
   197→   197→    return () => {
   198→   198→      if (debounceTimer.current) clearTimeout(debounceTimer.current)
   199→   199→    }
   200→   200→  }, [search])
   201→   201→
   202→   202→  // ── Data fetching effects ──
   203→   203→  useEffect(() => {
   204→   204→    fetchKelas()
   205→   205→  }, [fetchKelas])
   206→   206→
   207→   207→  useEffect(() => {
   208→   208→    fetchSiswa()
   209→   209→  }, [fetchSiswa])
   210→   210→
   211→   211→  // ── Helpers ──
   212→   212→  const getKelasNama = (kelasId: number) => {
   213→   213→    const k = kelasList.find((k) => k.id === kelasId)
   214→   214→    return k?.nama ?? kelasId.toString()
   215→   215→  }
   216→   216→
   217→   217→  const openAddModal = () => {
   218→   218→    setEditingId(null)
   219→   219→    setForm(EMPTY_FORM)
   220→   220→    setModalOpen(true)
   221→   221→  }
   222→   222→
   223→   223→  const openEditModal = (siswa: Siswa) => {
   224→   224→    setEditingId(siswa.id)
   225→   225→    setForm({
   226→   226→      nis: siswa.nis,
   227→   227→      nisn: siswa.nisn,
   228→   228→      nama: siswa.nama,
   229→   229→      jenisKelamin: siswa.jenisKelamin,
   230→   230→      tempatLahir: siswa.tempatLahir,
   231→   231→      tanggalLahir: siswa.tanggalLahir,
   232→   232→      agama: siswa.agama,
   233→   233→      alamat: siswa.alamat,
   234→   234→      namaAyah: siswa.namaAyah,
   235→   235→      namaIbu: siswa.namaIbu,
   236→   236→      noHp: siswa.noHp,
   237→   237→      kelasId: String(siswa.kelasId),
   238→   238→      status: siswa.status,
   239→   239→    })
   240→   240→    setModalOpen(true)
   241→   241→  }
   242→   242→
   243→   243→  const handleFormChange = (field: keyof FormData, value: string) => {
   244→   244→    setForm((prev) => ({ ...prev, [field]: value }))
   245→   245→  }
   246→   246→
   247→   247→  const handleSubmit = async () => {
   248→   248→    if (!form.nis.trim() || !form.nisn.trim() || !form.nama.trim()) {
   249→   249→      toast.error('NIS, NISN, dan Nama wajib diisi')
   250→   250→      return
   251→   251→    }
   252→   252→    if (!form.kelasId) {
   253→   253→      toast.error('Kelas wajib dipilih')
   254→   254→      return
   255→   255→    }
   256→   256→
   257→   257→    setFormSubmitting(true)
   258→   258→    try {
   259→   259→      const payload = {
   260→   260→        ...form,
   261→   261→        kelasId: Number(form.kelasId),
   262→   262→      }
   263→   263→      if (editingId) {
   264→   264→        await api.put('/siswa', { id: editingId, ...payload })
   265→   265→        toast.success('Data siswa berhasil diperbarui')
   266→   266→      } else {
   267→   267→        await api.post('/siswa', payload)
   268→   268→        toast.success('Data siswa berhasil ditambahkan')
   269→   269→      }
   270→   270→      setModalOpen(false)
   271→   271→      fetchSiswa()
   272→   272→    } catch (err) {
   273→   273→      toast.error(err instanceof Error ? err.message : 'Gagal menyimpan data')
   274→   274→    } finally {
   275→   275→      setFormSubmitting(false)
   276→   276→    }
   277→   277→  }
   278→   278→
   279→   279→  const confirmDelete = (siswa: Siswa) => {
   280→   280→    setDeletingItem(siswa)
   281→   281→    setDeleteOpen(true)
   282→   282→  }
   283→   283→
   284→   284→  const handleDelete = async () => {
   285→   285→    if (!deletingItem) return
   286→   286→    setDeleting(true)
   287→   287→    try {
   288→   288→      await api.del(`/siswa?id=${deletingItem.id}`)
   289→   289→      toast.success('Data siswa berhasil dihapus')
   290→   290→      setDeleteOpen(false)
   291→   291→      setDeletingItem(null)
   292→   292→      fetchSiswa()
   293→   293→    } catch (err) {
   294→   294→      toast.error(err instanceof Error ? err.message : 'Gagal menghapus data')
   295→   295→    } finally {
   296→   296→      setDeleting(false)
   297→   297→    }
   298→   298→  }
   299→   299→
   300→   300→  const handleImportExcel = () => {
   301→   301→    fileInputRef.current?.click()
   302→   302→  }
   303→   303→
   304→   304→  const handleFileChange = async (e: React.ChangeEvent<HTMLInputElement>) => {
   305→   305→    const file = e.target.files?.[0]
   306→   306→    if (!file) return
   307→   307→    const formData = new FormData()
   308→   308→    formData.append('file', file)
   309→   309→    try {
   310→   310→      const token = (await import('@/lib/store')).useAppStore.getState().token
   311→   311→      const headers: Record<string, string> = {}
   312→   312→      if (token) headers['Authorization'] = `Bearer ${token}`
   313→   313→      const res = await fetch('/api/siswa/import', {
   314→   314→        method: 'POST',
   315→   315→        headers,
   316→   316→        body: formData,
   317→   317→      })
   318→   318→      const json = await res.json()
   319→   319→      if (!res.ok) throw new Error(json.error || 'Gagal import')
   320→   320→      toast.success('Data berhasil diimport')
   321→   321→      fetchSiswa()
   322→   322→    } catch (err) {
   323→   323→      toast.error(err instanceof Error ? err.message : 'Gagal mengimport file')
   324→   324→    } finally {
   325→   325→      if (fileInputRef.current) fileInputRef.current.value = ''
   326→   326→    }
   327→   327→  }
   328→   328→
   329→   329→  const handleExportExcel = async () => {
   330→   330→    try {
   331→   331→      const token = (await import('@/lib/store')).useAppStore.getState().token
   332→   332→      const headers: Record<string, string> = {}
   333→   333→      if (token) headers['Authorization'] = `Bearer ${token}`
   334→   334→      const params = new URLSearchParams()
   335→   335→      if (debouncedSearch) params.set('search', debouncedSearch)
   336→   336→      if (filterKelas) params.set('kelas', filterKelas)
   337→   337→      const res = await fetch(`/api/siswa/export?${params.toString()}`, { headers })
   338→   338→      if (!res.ok) throw new Error('Gagal export')
   339→   339→      const blob = await res.blob()
   340→   340→      const url = URL.createObjectURL(blob)
   341→   341→      const a = document.createElement('a')
   342→   342→      a.href = url
   343→   343→      a.download = 'data-siswa.xlsx'
   344→   344→      a.click()
   345→   345→      URL.revokeObjectURL(url)
   346→   346→      toast.success('Data berhasil diexport')
   347→   347→    } catch (err) {
   348→   348→      toast.error(err instanceof Error ? err.message : 'Gagal mengexport data')
   349→   349→    }
   350→   350→  }
   351→   351→
   352→   352→  // ── Pagination range ──
   353→   353→  const startIdx = (page - 1) * PAGE_LIMIT
   354→   354→  const endIdx = Math.min(startIdx + PAGE_LIMIT, total)
   355→   355→
   356→   356→  return (
   357→   357→    <section className="w-full" style={{ backgroundColor: '#f4f6f9' }}>
   358→   358→      <div className="mx-auto max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   359→   359→        {/* ── Header ── */}
   360→   360→        <div className="mb-6">
   361→   361→          <h1
   362→   362→            className="text-2xl font-bold"
   363→   363→            style={{ color: '#0a2540' }}
   364→   364→          >
   365→   365→            Data Siswa
   366→   366→          </h1>
   367→   367→          <p className="mt-1 text-sm" style={{ color: '#0a2540' }}>
   368→   368→            Kelola data siswa sekolah dengan mudah
   369→   369→          </p>
   370→   370→        </div>
   371→   371→
   372→   372→        {/* ── Card ── */}
   373→   373→        <div className="rounded-2xl bg-white p-4 shadow-sm sm:p-6">
   374→   374→          {/* ── Action Bar ── */}
   375→   375→          <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   376→   376→            {/* Left: buttons */}
   377→   377→            <div className="flex flex-wrap items-center gap-2">
   378→   378→              <Button
   379→   379→                onClick={openAddModal}
   380→   380→                className="gap-1.5 text-white"
   381→   381→                style={{ backgroundColor: '#10b981' }}
   382→   382→              >
   383→   383→                <Plus className="size-4" />
   384→   384→                Tambah
   385→   385→              </Button>
   386→   386→              <Button
   387→   387→                variant="outline"
   388→   388→                onClick={handleImportExcel}
   389→   389→                className="gap-1.5"
   390→   390→                style={{ color: '#0a2540', borderColor: '#d1d5db' }}
   391→   391→              >
   392→   392→                <FileSpreadsheet className="size-4" />
   393→   393→                Import Excel
   394→   394→              </Button>
   395→   395→              <Button
   396→   396→                variant="outline"
   397→   397→                onClick={handleExportExcel}
   398→   398→                className="gap-1.5"
   399→   399→                style={{ color: '#0a2540', borderColor: '#d1d5db' }}
   400→   400→              >
   401→   401→                <Download className="size-4" />
   402→   402→                Export Excel
   403→   403→              </Button>
   404→   404→              <input
   405→   405→                ref={fileInputRef}
   406→   406→                type="file"
   407→   407→                accept=".xlsx,.xls"
   408→   408→                className="hidden"
   409→   409→                onChange={handleFileChange}
   410→   410→              />
   411→   411→            </div>
   412→   412→
   413→   413→            {/* Right: search + filter */}
   414→   414→            <div className="flex flex-wrap items-center gap-2">
   415→   415→              <div className="relative">
   416→   416→                <Search
   417→   417→                  className="absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-gray-400"
   418→   418→                />
   419→   419→                <Input
   420→   420→                  placeholder="Cari siswa..."
   421→   421→                  value={search}
   422→   422→                  onChange={(e) => setSearch(e.target.value)}
   423→   423→                  className="w-56 pl-9"
   424→   424→                />
   425→   425→              </div>
   426→   426→              <Select value={filterKelas} onValueChange={(v) => { setFilterKelas(v); setPage(1) }}>
   427→   427→                <SelectTrigger className="w-44">
   428→   428→                  <SelectValue placeholder="Semua Kelas" />
   429→   429→                </SelectTrigger>
   430→   430→                <SelectContent>
   431→   431→                  <SelectItem value="__all__">Semua Kelas</SelectItem>
   432→   432→                  {kelasList.map((k) => (
   433→   433→                    <SelectItem key={k.id} value={String(k.id)}>
   434→   434→                      {k.nama}
   435→   435→                    </SelectItem>
   436→   436→                  ))}
   437→   437→                </SelectContent>
   438→   438→              </Select>
   439→   439→            </div>
   440→   440→          </div>
   441→   441→
   442→   442→          {/* ── Table ── */}
   443→   443→          <div className="overflow-x-auto">
   444→   444→            {loading ? (
   445→   445→              /* Skeleton */
   446→   446→              <div className="space-y-3">
   447→   447→                {Array.from({ length: 5 }).map((_, i) => (
   448→   448→                  <div key={i} className="flex items-center gap-4">
   449→   449→                    <Skeleton className="h-8 w-8" />
   450→   450→                    <Skeleton className="h-8 w-20" />
   451→   451→                    <Skeleton className="h-8 w-24" />
   452→   452→                    <Skeleton className="h-8 flex-1" />
   453→   453→                    <Skeleton className="h-8 w-12" />
   454→   454→                    <Skeleton className="h-8 w-20" />
   455→   455→                    <Skeleton className="h-8 w-16" />
   456→   456→                    <Skeleton className="h-8 w-20" />
   457→   457→                  </div>
   458→   458→                ))}
   459→   459→              </div>
   460→   460→            ) : siswaList.length === 0 ? (
   461→   461→              /* Empty state */
   462→   462→              <div className="flex flex-col items-center justify-center py-16 text-center">
   463→   463→                <div
   464→   464→                  className="mb-4 flex size-16 items-center justify-center rounded-full"
   465→   465→                  style={{ backgroundColor: '#f0fdf4' }}
   466→   466→                >
   467→   467→                  <Inbox className="size-8" style={{ color: '#10b981' }} />
   468→   468→                </div>
   469→   469→                <h3
   470→   470→                  className="text-base font-semibold"
   471→   471→                  style={{ color: '#0a2540' }}
   472→   472→                >
   473→   473→                  Data Kosong
   474→   474→                </h3>
   475→   475→                <p className="mt-1 max-w-xs text-sm text-gray-500">
   476→   476→                  {debouncedSearch || filterKelas
   477→   477→                    ? 'Tidak ditemukan siswa yang sesuai dengan filter.'
   478→   478→                    : 'Belum ada data siswa. Klik tombol Tambah untuk menambahkan siswa baru.'}
   479→   479→                </p>
   480→   480→              </div>
   481→   481→            ) : (
   482→   482→              <table className="siadak-table w-full text-sm">
   483→   483→                <thead>
   484→   484→                  <tr
   485→   485→                    className="border-b text-left text-xs font-semibold uppercase tracking-wide"
   486→   486→                    style={{ color: '#0a2540', backgroundColor: '#f9fafb' }}
   487→   487→                  >
