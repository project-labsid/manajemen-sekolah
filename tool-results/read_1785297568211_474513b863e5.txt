   117→   117→function formatDateDisplay(dateStr: string): string {
   118→   118→  try {
   119→   119→    const date = new Date(dateStr + 'T00:00:00')
   120→   120→    return format(date, 'EEEE, dd MMMM yyyy', { locale: localeId })
   121→   121→  } catch {
   122→   122→    return dateStr
   123→   123→  }

   126→   126→function getBrowserInfo(): string {
   127→   127→  if (typeof navigator === 'undefined') return ''
   128→   128→  const ua = navigator.userAgent
   129→   129→  if (ua.includes('Firefox/')) return 'Firefox ' + ua.split('Firefox/')[1]?.split(' ')[0]
   130→   130→  if (ua.includes('Edg/')) return 'Edge ' + ua.split('Edg/')[1]?.split(' ')[0]
   131→   131→  if (ua.includes('Chrome/')) return 'Chrome ' + ua.split('Chrome/')[1]?.split(' ')[0]
   132→   132→  if (ua.includes('Safari/')) return 'Safari ' + ua.split('Version/')[1]?.split(' ')[0]
   133→   133→  return 'Unknown'
   134→   134→}
   135→   135→
   136→   136→function getDeviceInfo(): string {
   137→   137→  if (typeof navigator === 'undefined') return ''
   138→   138→  const ua = navigator.userAgent
   139→   139→  if (/iPhone|iPad/.test(ua)) return 'iOS Device'
   140→   140→  if (/Android/.test(ua)) return 'Android Device'
   141→   141→  if (/Windows/.test(ua)) return 'Windows PC'
   142→   142→  if (/Mac/.test(ua)) return 'Mac'
   143→   143→  if (/Linux/.test(ua)) return 'Linux'
   144→   144→  return 'Unknown Device'
   145→   145→}

   184→   184→function calculateDuration(jamMasuk: string, jamPulang: string): string {
   185→   185→  if (!jamMasuk || !jamPulang) return '-'
   186→   186→  try {
   187→   187→    const [h1, m1] = jamMasuk.split(':').map(Number)
   188→   188→    const [h2, m2] = jamPulang.split(':').map(Number)
   189→   189→    const diffMs = (h2 * 60 + m2 - h1 * 60 - m1) * 60 * 1000
   190→   190→    if (diffMs <= 0) return '-'
   191→   191→    const hours = Math.floor(diffMs / (1000 * 60 * 60))
   192→   192→    const minutes = Math.floor((diffMs % (1000 * 60 * 60)) / (1000 * 60))
   193→   193→    if (hours > 0) return `${hours} jam ${minutes} menit`
   194→   194→    return `${minutes} menit`
   195→   195→  } catch {
   196→   196→    return '-'
   197→   197→  }
   198→   198→}
