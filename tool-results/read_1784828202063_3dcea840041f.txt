   440→                  <tr key={i} className="border-b border-gray-100">
   441→                    <td className="px-4 py-3">
   442→                      <Skeleton className="h-4 w-6" />
   443→                    </td>
   444→                    <td className="px-4 py-3">
   445→                      <Skeleton className="h-4 w-24" />
   446→                    </td>

   448→                      <Skeleton className="h-4 w-32" />
   449→                    </td>

   452→                    </td>

   465→                        <Skeleton className="size-8 rounded-md" />

   482→                    <td className="px-4 py-3 text-gray-500">
   483→                      {(page - 1) * LIMIT + idx + 1}

   487→                    </td>
