   129→   129→// ── Component ───────────────────────────────────────────────────────────────
   130→   130→
   131→   131→export default function DataSiswa() {
   132→   132→  // ── Table state ──
   133→   133→  const [siswaList, setSiswaList] = useState<Siswa[]>([])
   134→   134→  const [total, setTotal] = useState(0)
   135→   135→  const [totalPages, setTotalPages] = useState(1)
   136→   136→  const [page, setPage] = useState(1)
   137→   137→  const [search, setSearch] = useState('')
   138→   138→  const [debouncedSearch, setDebouncedSearch] = useState('')
   139→   139→  const [filterKelas, setFilterKelas] = useState('')
   140→   140→  const [loading, setLoading] = useState(true)
   141→   141→
   142→   142→  // ── Kelas dropdown data ──
   143→   143→  const [kelasList, setKelasList] = useState<Kelas[]>([])
   144→   144→
   145→   145→  // ── Modal state ──
   146→   146→  const [modalOpen, setModalOpen] = useState(false)
   147→   147→  const [editingId, setEditingId] = useState<string | null>(null)
   148→   148→  const [form, setForm] = useState<FormData>(EMPTY_FORM)
   149→   149→  const [formSubmitting, setFormSubmitting] = useState(false)
   150→   150→
   151→   151→  // ── Delete confirmation state ──
   152→   152→  const [deleteOpen, setDeleteOpen] = useState(false)
   153→   153→  const [deletingItem, setDeletingItem] = useState<Siswa | null>(null)
   154→   154→  const [deleting, setDeleting] = useState(false)
   155→   155→
   156→   156→  // ── Refs ──
   157→   157→  const fileInputRef = useRef<HTMLInputElement>(null)
   158→   158→  const debounceTimer = useRef<ReturnType<typeof setTimeout>>()
   159→   159→
   160→   160→  // ── Fetch kelas list ──
   161→   161→  const fetchKelas = useCallback(async () => {
   162→   162→    try {
   163→   163→      const res = await api.get<Kelas[]>('/kelas')
   164→   164→      setKelasList(Array.isArray(res) ? res : (res as unknown as { data: Kelas[] }).data ?? [])
   165→   165→    } catch {
   166→   166→      // silently fail, dropdown will just be empty
   167→   167→    }
   168→   168→  }, [])
   169→   169→
   170→   170→  // ── Fetch siswa list ──
   171→   171→  const fetchSiswa = useCallback(async () => {
   172→   172→    setLoading(true)
   173→   173→    try {
   174→   174→      const params = new URLSearchParams()
   175→   175→      params.set('page', String(page))
   176→   176→      params.set('limit', String(PAGE_LIMIT))
   177→   177→      if (debouncedSearch) params.set('search', debouncedSearch)
   178→   178→      if (filterKelas) params.set('kelas', filterKelas)
   179→   179→
   180→   180→      const res = await api.get<SiswaResponse>(`/siswa?${params.toString()}`)
   181→   181→      const data = res as SiswaResponse
   182→   182→      setSiswaList(data.data ?? [])
   183→   183→      setTotal(data.total ?? 0)
   184→   184→      setTotalPages(data.totalPages ?? 1)
   185→   185→    } catch {
   186→   186→      toast.error('Gagal memuat data siswa')
   187→   187→    } finally {
   188→   188→      setLoading(false)
   189→   189→    }
   190→   190→  }, [page, debouncedSearch, filterKelas])
   191→   191→
   192→   192→  // ── Debounced search ──
   193→   193→  useEffect(() => {
   194→   194→    if (debounceTimer.current) clearTimeout(debounceTimer.current)
   195→   195→    debounceTimer.current = setTimeout(() => {
   196→   196→      setDebouncedSearch(search)
   197→   197→      setPage(1)
   198→   198→    }, 400)
   199→   199→    return () => {
   200→   200→      if (debounceTimer.current) clearTimeout(debounceTimer.current)
   201→   201→    }
   202→   202→  }, [search])
   203→   203→
   204→   204→  // ── Data fetching effects ──
   205→   205→  useEffect(() => {
   206→   206→    fetchKelas()
   207→   207→  }, [fetchKelas])
   208→   208→
   209→   209→  useEffect(() => {
   210→   210→    fetchSiswa()
   211→   211→  }, [fetchSiswa])
   212→   212→
   213→   213→  // ── Helpers ──
   214→   214→  const getKelasNama = (kodeKelas: string) => {
   215→   215→    const k = kelasList.find((k) => k.kodeKelas === kodeKelas)
   216→   216→    return k?.namaKelas ?? kodeKelas
   217→   217→  }
   218→   218→
   219→   219→  const openAddModal = () => {
   220→   220→    setEditingId(null)
   221→   221→    setForm(EMPTY_FORM)
   222→   222→    setModalOpen(true)
   223→   223→  }
   224→   224→
   225→   225→  const openEditModal = (siswa: Siswa) => {
   226→   226→    setEditingId(siswa.id)
   227→   227→    setForm({
   228→   228→      nis: siswa.nis,
   229→   229→      nisn: siswa.nisn,
   230→   230→      nama: siswa.nama,
   231→   231→      jenisKelamin: siswa.jenisKelamin,
   232→   232→      tempatLahir: siswa.tempatLahir,
   233→   233→      tanggalLahir: siswa.tanggalLahir,
   234→   234→      agama: siswa.agama,
   235→   235→      alamat: siswa.alamat,
   236→   236→      namaAyah: siswa.namaAyah,
   237→   237→      namaIbu: siswa.namaIbu,
   238→   238→      noHp: siswa.noHp,
   239→   239→      kelas: siswa.kelas || '',
   240→   240→      status: siswa.status,
   241→   241→    })
   242→   242→    setModalOpen(true)
   243→   243→  }
   244→   244→
   245→   245→  const handleFormChange = (field: keyof FormData, value: string) => {
   246→   246→    setForm((prev) => ({ ...prev, [field]: value }))
   247→   247→  }
   248→   248→
   249→   249→  const handleSubmit = async () => {
   250→   250→    if (!form.nis.trim() || !form.nisn.trim() || !form.nama.trim()) {
   251→   251→      toast.error('NIS, NISN, dan Nama wajib diisi')
   252→   252→      return
   253→   253→    }
   254→   254→    if (!form.kelas) {
   255→   255→      toast.error('Kelas wajib dipilih')
   256→   256→      return
   257→   257→    }
   258→   258→
   259→   259→    setFormSubmitting(true)
   260→   260→    try {
   261→   261→      const payload = {
   262→   262→        nis: form.nis,
   263→   263→        nisn: form.nisn,
   264→   264→        nama: form.nama,
   265→   265→        jenisKelamin: form.jenisKelamin,
   266→   266→        tempatLahir: form.tempatLahir,
   267→   267→        tanggalLahir: form.tanggalLahir,
   268→   268→        agama: form.agama,
   269→   269→        alamat: form.alamat,
   270→   270→        namaAyah: form.namaAyah,
   271→   271→        namaIbu: form.namaIbu,
   272→   272→        noHp: form.noHp,
   273→   273→        kelas: form.kelas,
   274→   274→        status: form.status,
   275→   275→      }
   276→   276→      if (editingId) {
   277→   277→        await api.put('/siswa', { id: editingId, ...payload })
   278→   278→        toast.success('Data siswa berhasil diperbarui')
   279→   279→      } else {
   280→   280→        await api.post('/siswa', payload)
   281→   281→        toast.success('Data siswa berhasil ditambahkan')
   282→   282→      }
   283→   283→      setModalOpen(false)
   284→   284→      fetchSiswa()
   285→   285→    } catch (err) {
   286→   286→      toast.error(err instanceof Error ? err.message : 'Gagal menyimpan data')
   287→   287→    } finally {
   288→   288→      setFormSubmitting(false)
   289→   289→    }
   290→   290→  }
   291→   291→
   292→   292→  const confirmDelete = (siswa: Siswa) => {
   293→   293→    setDeletingItem(siswa)
   294→   294→    setDeleteOpen(true)
   295→   295→  }
   296→   296→
   297→   297→  const handleDelete = async () => {
   298→   298→    if (!deletingItem) return
   299→   299→    setDeleting(true)
   300→   300→    try {
   301→   301→      await api.del(`/siswa?id=${deletingItem.id}`)
   302→   302→      toast.success('Data siswa berhasil dihapus')
   303→   303→      setDeleteOpen(false)
   304→   304→      setDeletingItem(null)
   305→   305→      fetchSiswa()
   306→   306→    } catch (err) {
   307→   307→      toast.error(err instanceof Error ? err.message : 'Gagal menghapus data')
   308→   308→    } finally {
   309→   309→      setDeleting(false)
   310→   310→    }
   311→   311→  }
   312→   312→
   313→   313→  const handleImportExcel = () => {
   314→   314→    fileInputRef.current?.click()
   315→   315→  }
   316→   316→
   317→   317→  const handleFileChange = async (e: React.ChangeEvent<HTMLInputElement>) => {
   318→   318→    const file = e.target.files?.[0]
   319→   319→    if (!file) return
   320→   320→    const formData = new FormData()
   321→   321→    formData.append('file', file)
   322→   322→    try {
   323→   323→      const token = (await import('@/lib/store')).useAppStore.getState().token
   324→   324→      const headers: Record<string, string> = {}
   325→   325→      if (token) headers['Authorization'] = `Bearer ${token}`
   326→   326→      const res = await fetch('/api/siswa/import', {
   327→   327→        method: 'POST',
   328→   328→        headers,
   329→   329→        body: formData,
   330→   330→      })
   331→   331→      const json = await res.json()
   332→   332→      if (!res.ok) throw new Error(json.error || 'Gagal import')
   333→   333→      toast.success('Data berhasil diimport')
   334→   334→      fetchSiswa()
   335→   335→    } catch (err) {
   336→   336→      toast.error(err instanceof Error ? err.message : 'Gagal mengimport file')
   337→   337→    } finally {
   338→   338→      if (fileInputRef.current) fileInputRef.current.value = ''
   339→   339→    }
   340→   340→  }
   341→   341→
   342→   342→  const handleExportExcel = async () => {
   343→   343→    try {
   344→   344→      const token = (await import('@/lib/store')).useAppStore.getState().token
   345→   345→      const headers: Record<string, string> = {}
   346→   346→      if (token) headers['Authorization'] = `Bearer ${token}`
   347→   347→      const params = new URLSearchParams()
   348→   348→      if (debouncedSearch) params.set('search', debouncedSearch)
   349→   349→      if (filterKelas) params.set('kelas', filterKelas)
   350→   350→      const res = await fetch(`/api/siswa/export?${params.toString()}`, { headers })
   351→   351→      if (!res.ok) throw new Error('Gagal export')
   352→   352→      const blob = await res.blob()
   353→   353→      const url = URL.createObjectURL(blob)
   354→   354→      const a = document.createElement('a')
   355→   355→      a.href = url
   356→   356→      a.download = 'data-siswa.xlsx'
   357→   357→      a.click()
   358→   358→      URL.revokeObjectURL(url)
   359→   359→      toast.success('Data berhasil diexport')
   360→   360→    } catch (err) {
   361→   361→      toast.error(err instanceof Error ? err.message : 'Gagal mengexport data')
   362→   362→    }
   363→   363→  }
   364→   364→
   365→   365→  // ── Pagination range ──
   366→   366→  const startIdx = (page - 1) * PAGE_LIMIT
   367→   367→  const endIdx = Math.min(startIdx + PAGE_LIMIT, total)
   368→   368→
   369→   369→  return (
   370→   370→    <section className="w-full" style={{ backgroundColor: '#f4f6f9' }}>
   371→   371→      <div className="mx-auto max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   372→   372→        {/* ── Header ── */}
   373→   373→        <div className="mb-6">
   374→   374→          <h1
   375→   375→            className="text-2xl font-bold"
   376→   376→            style={{ color: '#0a2540' }}
   377→   377→          >
   378→   378→            Data Siswa
   379→   379→          </h1>
   380→   380→          <p className="mt-1 text-sm" style={{ color: '#0a2540' }}>
   381→   381→            Kelola data siswa sekolah dengan mudah
   382→   382→          </p>
   383→   383→        </div>
   384→   384→
   385→   385→        {/* ── Card ── */}
   386→   386→        <div className="rounded-2xl bg-white p-4 shadow-sm sm:p-6">
   387→   387→          {/* ── Action Bar ── */}
   388→   388→          <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   389→   389→            {/* Left: buttons */}
   390→   390→            <div className="flex flex-wrap items-center gap-2">
   391→   391→              <Button
   392→   392→                onClick={openAddModal}
   393→   393→                className="gap-1.5 text-white"
   394→   394→                style={{ backgroundColor: '#10b981' }}
   395→   395→              >
   396→   396→                <Plus className="size-4" />
   397→   397→                Tambah
   398→   398→              </Button>
   399→   399→              <Button
   400→   400→                variant="outline"
   401→   401→                onClick={handleImportExcel}
   402→   402→                className="gap-1.5"
   403→   403→                style={{ color: '#0a2540', borderColor: '#d1d5db' }}
   404→   404→              >
   405→   405→                <FileSpreadsheet className="size-4" />
   406→   406→                Import Excel
   407→   407→              </Button>
   408→   408→              <Button
   409→   409→                variant="outline"
   410→   410→                onClick={handleExportExcel}
   411→   411→                className="gap-1.5"
   412→   412→                style={{ color: '#0a2540', borderColor: '#d1d5db' }}
   413→   413→              >
   414→   414→                <Download className="size-4" />
   415→   415→                Export Excel
   416→   416→              </Button>
   417→   417→              <input
   418→   418→                ref={fileInputRef}
   419→   419→                type="file"
   420→   420→                accept=".xlsx,.xls"
   421→   421→                className="hidden"
   422→   422→                onChange={handleFileChange}
   423→   423→              />
   424→   424→            </div>
   425→   425→
   426→   426→            {/* Right: search + filter */}
   427→   427→            <div className="flex flex-wrap items-center gap-2">
   428→   428→              <div className="relative">
   429→   429→                <Search
   430→   430→                  className="absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-gray-400"
   431→   431→                />
   432→   432→                <Input
   433→   433→                  placeholder="Cari siswa..."
   434→   434→                  value={search}
   435→   435→                  onChange={(e) => setSearch(e.target.value)}
   436→   436→                  className="w-56 pl-9"
   437→   437→                />
   438→   438→              </div>
   439→   439→              <Select value={filterKelas} onValueChange={(v) => { setFilterKelas(v); setPage(1) }}>
   440→   440→                <SelectTrigger className="w-44">
   441→   441→                  <SelectValue placeholder="Semua Kelas" />
   442→   442→                </SelectTrigger>
   443→   443→                <SelectContent>
   444→   444→                  <SelectItem value="__all__">Semua Kelas</SelectItem>
   445→   445→                  {kelasList.map((k) => (
   446→   446→                    <SelectItem key={k.kodeKelas} value={k.kodeKelas}>
   447→   447→                      {k.namaKelas}
   448→   448→                    </SelectItem>
   449→   449→                  ))}
   450→   450→                </SelectContent>
   451→   451→              </Select>
   452→   452→            </div>
   453→   453→          </div>
   454→   454→
   455→   455→          {/* ── Table ── */}
   456→   456→          <div className="overflow-x-auto">
   457→   457→            {loading ? (
   458→   458→              /* Skeleton */
   459→   459→              <div className="space-y-3">
   460→   460→                {Array.from({ length: 5 }).map((_, i) => (
   461→   461→                  <div key={i} className="flex items-center gap-4">
   462→   462→                    <Skeleton className="h-8 w-8" />
   463→   463→                    <Skeleton className="h-8 w-20" />
   464→   464→                    <Skeleton className="h-8 w-24" />
   465→   465→                    <Skeleton className="h-8 flex-1" />
   466→   466→                    <Skeleton className="h-8 w-12" />
   467→   467→                    <Skeleton className="h-8 w-20" />
   468→   468→                    <Skeleton className="h-8 w-16" />
   469→   469→                    <Skeleton className="h-8 w-20" />
   470→   470→                  </div>
   471→   471→                ))}
   472→   472→              </div>
   473→   473→            ) : siswaList.length === 0 ? (
   474→   474→              /* Empty state */
   475→   475→              <div className="flex flex-col items-center justify-center py-16 text-center">
   476→   476→                <div
   477→   477→                  className="mb-4 flex size-16 items-center justify-center rounded-full"
   478→   478→                  style={{ backgroundColor: '#f0fdf4' }}
   479→   479→                >
   480→   480→                  <Inbox className="size-8" style={{ color: '#10b981' }} />
   481→   481→                </div>
   482→   482→                <h3
   483→   483→                  className="text-base font-semibold"
   484→   484→                  style={{ color: '#0a2540' }}
   485→   485→                >
   486→   486→                  Data Kosong
   487→   487→                </h3>
   488→   488→                <p className="mt-1 max-w-xs text-sm text-gray-500">
