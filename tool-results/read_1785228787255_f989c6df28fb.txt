    76→function getTodayString(): string {
    77→  return new Date().toISOString().split('T')[0]
    78→}
    79→
    80→function formatDateDisplay(dateStr: string): string {
    81→  try {
    82→    const date = new Date(dateStr + 'T00:00:00')
    83→    return format(date, 'EEEE, dd MMMM yyyy', { locale: localeId })
    84→  } catch {
    85→    return dateStr
    86→  }
    87→}
    88→
    89→function getBrowserInfo(): string {
    90→  if (typeof navigator === 'undefined') return ''
    91→  const ua = navigator.userAgent
    92→  if (ua.includes('Firefox/')) return 'Firefox ' + ua.split('Firefox/')[1]?.split(' ')[0]
    93→  if (ua.includes('Edg/')) return 'Edge ' + ua.split('Edg/')[1]?.split(' ')[0]
    94→  if (ua.includes('Chrome/')) return 'Chrome ' + ua.split('Chrome/')[1]?.split(' ')[0]
    95→  if (ua.includes('Safari/')) return 'Safari ' + ua.split('Version/')[1]?.split(' ')[0]
    96→  return 'Unknown'
    97→}
    98→
    99→function getDeviceInfo(): string {
   100→  if (typeof navigator === 'undefined') return ''
   101→  const ua = navigator.userAgent
   102→  if (/iPhone|iPad/.test(ua)) return 'iOS Device'
   103→  if (/Android/.test(ua)) return 'Android Device'
   104→  if (/Windows/.test(ua)) return 'Windows PC'
   105→  if (/Mac/.test(ua)) return 'Mac'
   106→  if (/Linux/.test(ua)) return 'Linux'
   107→  return 'Unknown Device'
   108→}
   109→
   110→function getLocalTime(): string {
   111→  const now = new Date()

   115→function getStatusBadge(status: string) {
   116→  switch (status) {
   117→    case 'Hadir':
   118→      return (
   119→        <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-0">

   136→        </Badge>
   137→      )
   138→    default:
   139→      return (
   140→        <Badge className="bg-gray-100 text-gray-600 hover:bg-gray-100 border-0">
   141→          {status || '-'}
   142→        </Badge>
   143→      )
   144→  }
   145→}
   146→
   147→function calculateDuration(jamMasuk: string, jamPulang: string): string {
   148→  if (!jamMasuk || !jamPulang) return '-'
   149→  try {
   150→    const [h1, m1] = jamMasuk.split(':').map(Number)
   151→    const [h2, m2] = jamPulang.split(':').map(Number)
   152→    const diffMs = (h2 * 60 + m2 - h1 * 60 - m1) * 60 * 1000
   153→    if (diffMs <= 0) return '-'
   154→    const hours = Math.floor(diffMs / (1000 * 60 * 60))
   155→    const minutes = Math.floor((diffMs % (1000 * 60 * 60)) / (1000 * 60))
   156→    if (hours > 0) return `${hours} jam ${minutes} menit`
   157→    return `${minutes} menit`
   158→  } catch {
   159→    return '-'
   160→  }
   161→}
