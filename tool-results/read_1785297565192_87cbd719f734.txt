   113→function getTodayString(): string {
   114→  return new Date().toISOString().split('T')[0]
   115→}
   116→
   117→function formatDateDisplay(dateStr: string): string {
   118→  try {
   119→    const date = new Date(dateStr + 'T00:00:00')
   120→    return format(date, 'EEEE, dd MMMM yyyy', { locale: localeId })
   121→  } catch {
   122→    return dateStr
   123→  }
   124→}
   125→
   126→function getBrowserInfo(): string {
   127→  if (typeof navigator === 'undefined') return ''
   128→  const ua = navigator.userAgent
   129→  if (ua.includes('Firefox/')) return 'Firefox ' + ua.split('Firefox/')[1]?.split(' ')[0]
   130→  if (ua.includes('Edg/')) return 'Edge ' + ua.split('Edg/')[1]?.split(' ')[0]
   131→  if (ua.includes('Chrome/')) return 'Chrome ' + ua.split('Chrome/')[1]?.split(' ')[0]
   132→  if (ua.includes('Safari/')) return 'Safari ' + ua.split('Version/')[1]?.split(' ')[0]
   133→  return 'Unknown'
   134→}
   135→
   136→function getDeviceInfo(): string {
   137→  if (typeof navigator === 'undefined') return ''
   138→  const ua = navigator.userAgent
   139→  if (/iPhone|iPad/.test(ua)) return 'iOS Device'
   140→  if (/Android/.test(ua)) return 'Android Device'
   141→  if (/Windows/.test(ua)) return 'Windows PC'
   142→  if (/Mac/.test(ua)) return 'Mac'
   143→  if (/Linux/.test(ua)) return 'Linux'
   144→  return 'Unknown Device'
   145→}

   179→        </Badge>

   183→
   184→function calculateDuration(jamMasuk: string, jamPulang: string): string {
   185→  if (!jamMasuk || !jamPulang) return '-'
   186→  try {
   187→    const [h1, m1] = jamMasuk.split(':').map(Number)
   188→    const [h2, m2] = jamPulang.split(':').map(Number)
   189→    const diffMs = (h2 * 60 + m2 - h1 * 60 - m1) * 60 * 1000
   190→    if (diffMs <= 0) return '-'
   191→    const hours = Math.floor(diffMs / (1000 * 60 * 60))
   192→    const minutes = Math.floor((diffMs % (1000 * 60 * 60)) / (1000 * 60))
   193→    if (hours > 0) return `${hours} jam ${minutes} menit`
   194→    return `${minutes} menit`
   195→  } catch {
   196→    return '-'
   197→  }
   198→}

   215→  const [submittingPulang, setSubmittingPulang] = useState(false)
   216→
   217→  // ── Pulang dialog ──
   218→  const [pulangDialogOpen, setPulangDialogOpen] = useState(false)
   219→  const [pulangKeterangan, setPulangKeterangan] = useState('')

   223→  // ── Sakit/Izin dialog ──
   224→  const [sakitIzinDialogOpen, setSakitIzinDialogOpen] = useState(false)

   226→  const [sakitIzinKeterangan, setSakitIzinKeterangan] = useState('')

   228→
   229→  // ── Compute: today string for the selected date ──
   230→  const selectedDateStr = useMemo(
   231→    () => format(selectedDate, 'yyyy-MM-dd'),
   232→    [selectedDate],
   233→  )
   234→  const isToday = selectedDateStr === getTodayString()
   235→
   236→  // ── Compute: own today's record (exclude fake pending entries) ──
   237→  const myTodayRecord = useMemo(() => {
   238→    if (isRekapViewer && !isToday) return null
   239→    return records.find(
   240→      (r) => r.namaGuru === user?.nama && !r.id.startsWith('pending-'),
   241→    ) || null
   242→  }, [records, user, isRekapViewer, isToday])
   243→
   244→  // ── Compute: own info cards data ──
   245→  const ownJamMasuk = myTodayRecord?.jamMasuk || '-'
   246→  const ownJamPulang = myTodayRecord?.jamPulang || '-'
   247→  const ownDurasi = myTodayRecord?.durasi ||
   248→    (myTodayRecord?.jamMasuk && !myTodayRecord?.jamPulang
   249→      ? 'Masih bekerja'
   250→      : '-')
   251→  const ownStatus = myTodayRecord?.status || 'Tidak Hadir'
   252→
   253→  // ── Can absen masuk? (only on today, no existing record) ──
   254→  const canAbsenMasuk = isToday && !myTodayRecord
   255→  // ── Can absen pulang? (today, has record, not yet clocked out) ──
   256→  const canAbsenPulang = isToday && myTodayRecord && !myTodayRecord.jamPulang
   257→  // ── Can sakit/izin? (today, no record yet) ──
   258→  const canSakitIzin = isToday && !myTodayRecord
   259→
   260→  // ── Geolocation ──
   261→  const getGeolocation = useCallback((): Promise<{ latitude: string; longitude: string }> => {
   262→    return new Promise((resolve) => {
   263→      if (typeof navigator === 'undefined' || !navigator.geolocation) {
   264→        resolve({ latitude: '', longitude: '' })
   265→        return
   266→      }
   267→      navigator.geolocation.getCurrentPosition(
   268→        (pos) => {
   269→          resolve({
   270→            latitude: String(pos.coords.latitude),
   271→            longitude: String(pos.coords.longitude),
   272→          })
   273→        },
   274→        () => {
   275→          resolve({ latitude: '', longitude: '' })
   276→        },
   277→        { enableHighAccuracy: false, timeout: 5000 },
   278→      )
   279→    })
   280→  }, [])
   281→
   282→  // ── Fetch attendance records ──
   283→  const fetchRecords = useCallback(async () => {
   284→    setLoading(true)
   285→    try {
   286→      const params = new URLSearchParams({ tanggal: selectedDateStr })
   287→      const res = await api.get<AbsensiResponse>(`/absensi-guru?${params.toString()}`)
   288→      const absensiRes = res as unknown as AbsensiResponse & { isRekap?: boolean }
   289→      const data = Array.isArray(absensiRes.data) ? absensiRes.data : []
   290→
   291→      if (isRekapViewer) {
   292→        // Admin/rekap: API already returns merged data (all staff + attendance status)
   293→        setRecords(data)
   294→      } else {
   295→        // Non-admin: only show own record
   296→        const filtered = data.filter((r) => r.namaGuru === user?.nama)
   297→        setRecords(filtered)
   298→      }
   299→    } catch {
   300→      toast.error('Gagal memuat data absensi')
   301→      setRecords([])
   302→    } finally {
   303→      setLoading(false)
   304→    }
   305→  }, [selectedDateStr, isRekapViewer, user])
   306→
   307→  useEffect(() => {
   308→    fetchRecords()
   309→  }, [fetchRecords])
   310→
   311→  // ── Handle Absen Masuk ──
   312→  const handleAbsenMasuk = useCallback(async () => {
   313→    if (!user) return
   314→
   315→    setSubmittingMasuk(true)
   316→    try {
   317→      const geo = await getGeolocation()
   318→      const browser = getBrowserInfo()
   319→      const device = getDeviceInfo()
   320→      const jamMasuk = getLocalTime()
   321→
   322→      await api.post('/absensi-guru', {
   323→        namaGuru: user.nama,
   324→        nip: user.nip || user.username || '',
   325→        jamMasuk,
   326→        latitude: geo.latitude,
   327→        longitude: geo.longitude,
   328→        browser,
   329→        device,
   330→        keterangan: '',
   331→      })
   332→
   333→      toast.success('Absen masuk berhasil dicatat!')
   334→      fetchRecords()
   335→    } catch (err: unknown) {
   336→      const message = err instanceof Error ? err.message : 'Gagal melakukan absen masuk'
   337→      toast.error(message)
   338→    } finally {
   339→      setSubmittingMasuk(false)
   340→    }
   341→  }, [user, getGeolocation, fetchRecords])
   342→
   343→  // ── Handle Absen Pulang: open dialog ──
   344→  const handleOpenPulangDialog = useCallback(() => {
   345→    if (!myTodayRecord || myTodayRecord.id.startsWith('pending-')) return
   346→    setPulangRecordId(myTodayRecord.id)
   347→    setPulangKeterangan('')
   348→    setPulangJam(getLocalTime())
   349→    setPulangDialogOpen(true)
   350→  }, [myTodayRecord])
   351→
   352→  // ── Handle Absen Pulang: submit ──
   353→  const handleAbsenPulang = useCallback(async () => {
   354→    if (!pulangRecordId || pulangRecordId.startsWith('pending-')) return
   355→
   356→    setSubmittingPulang(true)
   357→    try {
   358→      await api.put('/absensi-guru', {
   359→        id: pulangRecordId,
   360→        jamPulang: pulangJam,
   361→        keterangan: pulangKeterangan,
   362→      })
   363→
   364→      toast.success('Absen pulang berhasil dicatat!')
   365→      setPulangDialogOpen(false)
   366→      setPulangKeterangan('')
   367→      setPulangRecordId(null)
   368→      setPulangJam('')
   369→      fetchRecords()
   370→    } catch (err: unknown) {
   371→      const message = err instanceof Error ? err.message : 'Gagal melakukan absen pulang'
   372→      toast.error(message)
   373→    } finally {
   374→      setSubmittingPulang(false)
   375→    }
   376→  }, [pulangRecordId, pulangKeterangan, fetchRecords])
   377→
   378→  // ── Handle Sakit/Izin Submit ──
   379→  const handleSakitIzin = useCallback(async () => {
   380→    if (!user) return
   381→    if (!sakitIzinKeterangan.trim()) {
   382→      toast.error('Keterangan wajib diisi untuk Sakit/Izin')
   383→      return
   384→    }
   385→
   386→    setSubmittingSakitIzin(true)
   387→    try {
   388→      await api.post('/absensi-guru', {
   389→        namaGuru: user.nama,
   390→        nip: user.nip || user.username || '',
   391→        status: sakitIzinStatus,
   392→        keterangan: sakitIzinKeterangan.trim(),
   393→      })
   394→
   395→      toast.success(`Absen ${sakitIzinStatus.toLowerCase()} berhasil dicatat!`)
   396→      setSakitIzinDialogOpen(false)
   397→      setSakitIzinKeterangan('')
   398→      fetchRecords()
   399→    } catch (err: unknown) {
   400→      const message = err instanceof Error ? err.message : `Gagal melakukan absen ${sakitIzinStatus.toLowerCase()}`
   401→      toast.error(message)
   402→    } finally {
   403→      setSubmittingSakitIzin(false)
   404→    }

   415→            type="button"

   418→            className={
   419→              'relative flex items-center justify-center gap-3 rounded-2xl p-5 sm:p-6 text-white font-semibold text-base transition-all shadow-sm '
   420→              +
   421→              (canAbsenMasuk && !submittingMasuk
   422→                ? 'bg-[#10b981] hover:bg-[#059669] hover:shadow-md active:scale-[0.98]'

   429→              <LogIn className="h-6 w-6" />

   434→                Sudah diabsen

   437→          </button>

   439→          {/* Absen Pulang Button */}
   440→          <button
   441→            type="button"
   442→            disabled={!canAbsenPulang || submittingPulang}
   443→            onClick={handleOpenPulangDialog}
   444→            className={
   445→              'relative flex items-center justify-center gap-3 rounded-2xl p-5 sm:p-6 text-white font-semibold text-base transition-all shadow-sm '
   446→              +
   447→              (canAbsenPulang && !submittingPulang

   449→                : 'bg-gray-300 cursor-not-allowed opacity-60')

   459→              <span className="absolute top-2 right-3 text-xs font-normal opacity-80">
   460→                Sudah pulang

   463→          </button>

   465→          {/* Sakit/Izin Button */}
