    80→    80→  ph2: string

    92→    92→function parseNum(val: string): number {
    93→    93→  const n = parseFloat(val)
    94→    94→  return isNaN(n) ? 0 : n
    95→    95→}
    96→    96→
    97→    97→function clampScore(val: string): string {
    98→    98→  const n = parseFloat(val)
    99→    99→  if (val === '' || isNaN(n)) return ''
   100→   100→  if (n < 0) return '0'
   101→   101→  if (n > 100) return '100'
   102→   102→  return String(Math.round(n))
   103→   103→}
   104→   104→
   105→   105→function hitungRataRata(ph1: number, ph2: number, ph3: number, ph4: number): number {
   106→   106→  return Math.round(((ph1 + ph2 + ph3 + ph4) / 4) * 100) / 100
   107→   107→}
   108→   108→
   109→   109→function hitungNilaiAkhir(rataRata: number, pts: number, pas: number): number {
   110→   110→  return Math.round((rataRata * 0.4 + pts * 0.3 + pas * 0.3) * 100) / 100
   111→   111→}
   112→   112→
   113→   113→function hitungPredikat(nilaiAkhir: number): string {
   114→   114→  if (nilaiAkhir >= 90) return 'A'
   115→   115→  if (nilaiAkhir >= 80) return 'B+'
   116→   116→  if (nilaiAkhir >= 70) return 'B'
   117→   117→  if (nilaiAkhir >= 60) return 'C'
   118→   118→  return 'D'
   119→   119→}
   120→   120→
   121→   121→function predikatBadgeClass(predikat: string): string {
   122→   122→  switch (predikat) {
   123→   123→    case 'A':
   124→   124→      return 'bg-green-100 text-green-700 border-green-200'
   125→   125→    case 'B+':
   126→   126→      return 'bg-blue-100 text-blue-700 border-blue-200'
   127→   127→    case 'B':
   128→   128→      return 'bg-yellow-100 text-yellow-700 border-yellow-200'
   129→   129→    case 'C':
   130→   130→      return 'bg-orange-100 text-orange-700 border-orange-200'
   131→   131→    case 'D':
   132→   132→      return 'bg-red-100 text-red-700 border-red-200'
   133→   133→    default:
   134→   134→      return 'bg-gray-100 text-gray-500 border-gray-200'
   135→   135→  }
   136→   136→}
