    80→    80→function formatDateDisplay(dateStr: string): string {
    81→    81→  try {
    82→    82→    const date = new Date(dateStr + 'T00:00:00')
    83→    83→    return format(date, 'EEEE, dd MMMM yyyy', { locale: localeId })
    84→    84→  } catch {
    85→    85→    return dateStr
    86→    86→  }

    89→    89→function getBrowserInfo(): string {
    90→    90→  if (typeof navigator === 'undefined') return ''
    91→    91→  const ua = navigator.userAgent
    92→    92→  if (ua.includes('Firefox/')) return 'Firefox ' + ua.split('Firefox/')[1]?.split(' ')[0]
    93→    93→  if (ua.includes('Edg/')) return 'Edge ' + ua.split('Edg/')[1]?.split(' ')[0]
    94→    94→  if (ua.includes('Chrome/')) return 'Chrome ' + ua.split('Chrome/')[1]?.split(' ')[0]
    95→    95→  if (ua.includes('Safari/')) return 'Safari ' + ua.split('Version/')[1]?.split(' ')[0]
    96→    96→  return 'Unknown'
    97→    97→}
    98→    98→
    99→    99→function getDeviceInfo(): string {
   100→   100→  if (typeof navigator === 'undefined') return ''
   101→   101→  const ua = navigator.userAgent
   102→   102→  if (/iPhone|iPad/.test(ua)) return 'iOS Device'
   103→   103→  if (/Android/.test(ua)) return 'Android Device'
   104→   104→  if (/Windows/.test(ua)) return 'Windows PC'
   105→   105→  if (/Mac/.test(ua)) return 'Mac'
   106→   106→  if (/Linux/.test(ua)) return 'Linux'
   107→   107→  return 'Unknown Device'
   108→   108→}

   110→   110→function getLocalTime(): string {

   147→   147→function calculateDuration(jamMasuk: string, jamPulang: string): string {
   148→   148→  if (!jamMasuk || !jamPulang) return '-'
   149→   149→  try {
   150→   150→    const [h1, m1] = jamMasuk.split(':').map(Number)
   151→   151→    const [h2, m2] = jamPulang.split(':').map(Number)
   152→   152→    const diffMs = (h2 * 60 + m2 - h1 * 60 - m1) * 60 * 1000
   153→   153→    if (diffMs <= 0) return '-'
   154→   154→    const hours = Math.floor(diffMs / (1000 * 60 * 60))
   155→   155→    const minutes = Math.floor((diffMs % (1000 * 60 * 60)) / (1000 * 60))
   156→   156→    if (hours > 0) return `${hours} jam ${minutes} menit`
   157→   157→    return `${minutes} menit`
   158→   158→  } catch {
   159→   159→    return '-'
   160→   160→  }
   161→   161→}
